% AAAAT Blueprint contract: formatting/rendering commands shared by standalone and combined documents.
\NewDocumentCommand{\AAAATApplyCvLayout}{}
  {
    \setlength{\parskip}{0.25em}
    \setlength{\oddsidemargin}{-0.2in}
    \setlength{\evensidemargin}{-0.2in}
    \setlength{\textwidth}{6.9in}
    \setlength{\topmargin}{-0.55in}
    \setlength{\textheight}{9.65in}
  }

\NewDocumentCommand{\AAAATApplyLetterLayout}{}
  {
    \setlength{\parskip}{0.8em}
    \setlength{\oddsidemargin}{0in}
    \setlength{\evensidemargin}{0in}
    \setlength{\textwidth}{6.5in}
    \setlength{\topmargin}{-0.35in}
    \setlength{\textheight}{9.25in}
  }

\NewDocumentCommand{\AAAATRenderCvBlueprint}{}
  {
    \AAAATSelectDocumentLanguage
    \AAAATFullWidthHeader{\AAAATRenderDocumentTitle}
    \AAAATTwoRegionBody
      {\AAAATRenderSecondaryBlocks}
      {\AAAATRenderMainBlocks}
  }

\NewDocumentCommand{\AAAATRenderLetterBlueprint}{}
  {
    \AAAATSelectDocumentLanguage
    \AAAATRenderLetter
  }
